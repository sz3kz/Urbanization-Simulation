@startuml
skinparam style strictuml

participant "Main" as M
participant "sim : Simulation" as S
participant "previous_board : BoardOccupants" as Prev
participant "probability_board : BoardProbabilities" as Prob
participant "cell : CellOccupant" as Cell
participant "building : Building" as Bld
participant "next_board : BoardOccupants" as Next

M -> S : run()
activate S

loop forever

    note over S
    STEP 1:
    propagateBuildingProbabilities()
    end note

    loop for every occupied cell
        S -> Prev : getCellAtCoordinates()
        Prev --> S : CellOccupant

        S -> Cell : getBuilding()
        Cell --> S : Building*

        alt building exists
            S -> Bld : applyProbabilities(callbacks...)
            activate Bld

            note right
            Building queries neighbouring cells
            through callbacks supplied
            by Simulation
            end note

            Bld -> S : askCellExists(...)
            S --> Bld : bool

            Bld -> S : askCellEmpty(...)
            S --> Bld : bool

            Bld -> S : setProbability(...)
            S -> Prob : setProbabilityTypePercentageAtCoordinates(...)

            deactivate Bld
        end
    end

    note over S
    STEP 2:
    executeProbability()
    end note

    loop for every cell
        S -> Prob : getProbability(...)
        Prob --> S : probability

        S -> S : rollProbabilityDice()

        alt event triggered
            S -> Next : acquireOccupantToCoordinates()
        else
            S -> Next : copy/release occupant
        end
    end

    note over S
    STEP 3:
    decayBuildings()
    end note

    loop for every occupied cell
        S -> Cell : getBuilding()
        Cell --> S : Building*

        S -> Bld : decay()

        alt TTL exhausted
            S -> Cell : transformState()
        end
    end

    note over S
    STEP 4:
    recycleBoards()
    end note

    S -> S : swap(previous_board,next_board)

    note over S
    STEP 5:
    resetProbabilities()
    end note

    S -> Prob : resetProbabilities(previous_board)

    note over S
    STEP 6:
    incrementIteration()
    end note

    S -> S : current_iteration++

end

deactivate S
@enduml
