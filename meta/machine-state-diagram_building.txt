@startuml
title Building Lifecycle State Machine

[*] --> Normal : Building created

state "NORMAL" as Normal {
  Normal : BuildingState::NORMAL
  Normal : applyProbabilities() active
  Normal : TTL = 3700
}

state "BURNING" as Burning {
  Burning : BuildingState::BURNING
  Burning : Can ignite neighbours
  Burning : TTL = 200
}

state "RUIN" as Ruin {
  Ruin : BuildingState::RUIN
  Ruin : Limited influence
  Ruin : TTL = 560
}

state "Removed" as Empty {
  Empty : occupant == nullptr
}

Normal --> Burning : SET_CURRENT_BUILDING_ON_FIRE

Normal --> Ruin : TTL <= decay threshold : transformState()

Burning --> Ruin : TTL <= decay threshold : transformState()

Ruin --> Normal : RESTORE_FROM_RUIN

Ruin --> Empty : TTL <= decay threshold : transformState()

Empty --> Normal : CREATE_NEW_HOUSE : CREATE_NEW_SHOP : CREATE_NEW_FACTORY : CREATE_NEW_FIRESTATION : CREATE_NEW_CHURCH

@enduml
